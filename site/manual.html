public class DatabaseHelper {

    private static final Logger LOG = Logger.getLogger(DatabaseHelper.class.getName());

    /** วาง bonus_suki.db ไว้ที่รากโปรเจกต์ (โฟลเดอร์เดียวกับ pom.xml) */
    public static final String DB_FILE = "bonus_suki.db";
    private static final String URL = "jdbc:sqlite:" + DB_FILE;

    private static Connection conn;     // ใช้ร่วมกันทั้งแอป ห้าม DAO ปิดเอง

    private DatabaseHelper() {
    }

    public static synchronized Connection getConnect() {
        if (conn == null) {
            try {
                conn = DriverManager.getConnection(URL);
                try (Statement pragma = conn.createStatement()) {
                    pragma.execute("PRAGMA foreign_keys = ON");   // SQLite ปิด FK ไว้เป็นค่าเริ่มต้น
                }
            } catch (SQLException ex) {
                LOG.log(Level.SEVERE, "เชื่อมต่อฐานข้อมูลไม่สำเร็จ: " + URL, ex);
            }
        }
        return conn;
    }

    public static synchronized void close() {
        if (conn != null) {
            try {
                conn.close();
                conn = null;
            } catch (SQLException ex) {
                LOG.log(Level.WARNING, "ปิด connection ไม่สำเร็จ", ex);
            }
        }
    }

    /**
     * สร้างรหัสถัดไปจากรหัสล่าสุดในตาราง
     * nextId("Member", "Member_ID", "M", 3)  → M006
     * nextId("Bill", "Order_ID", "B", 4)     → B0007
     */
    public static String nextId(String table, String idColumn, String prefix, int digits) {
        String sql = "SELECT " + idColumn + " FROM " + table
                + " WHERE " + idColumn + " LIKE ?"
                + " ORDER BY LENGTH(" + idColumn + ") DESC, " + idColumn + " DESC LIMIT 1";
        int next = 1;
        try (PreparedStatement stmt = getConnect().prepareStatement(sql)) {
            stmt.setString(1, prefix + "%");
            try (ResultSet rs = stmt.executeQuery()) {
                if (rs.next()) {
                    next = Integer.parseInt(rs.getString(1).substring(prefix.length())) + 1;
                }
            }
        } catch (SQLException | NumberFormatException ex) {
            LOG.log(Level.WARNING, "nextId(" + table + ") ล้มเหลว", ex);
        }
        return prefix + String.format("%0" + digits + "d", next);
    }

    /** คอลัมน์ INT ที่ NULL ได้ (เช่น Promo_Amount_Member) → Integer หรือ null */
    public static Integer getNullableInt(ResultSet rs, String column) throws SQLException {
        int value = rs.getInt(column);
        return rs.wasNull() ? null : value;
    }

    /** คอลัมน์ DECIMAL ที่ NULL ได้ (เช่น Received_Amount) → Double หรือ null */
    public static Double getNullableDouble(ResultSet rs, String column) throws SQLException {
        double value = rs.getDouble(column);
        return rs.wasNull() ? null : value;
    }

package com.example.bonussuki.service;

import com.example.bonussuki.dao.MemberDao;
import com.example.bonussuki.model.Member;
import java.util.List;
import java.util.stream.Collectors;

public class MemberService {

    /** ทีมกำหนดเอง (รูปเล่มไม่ได้ระบุ): ยอดสุทธิทุก 25 บาท = 1 คะแนน */
    public static final int BAHT_PER_POINT = 25;

    private final MemberDao memberDao = new MemberDao();

    public List<Member> getAll() {
        return memberDao.getAll();
    }

    /** ค้นในหน่วยความจำ ไม่ต่อข้อความผู้ใช้เข้า SQL */
    public List<Member> search(String keyword) {
        String k = keyword.trim();
        return memberDao.getAll().stream()
                .filter(m -> m.getMemberName().contains(k) || m.getMemberPhoneNo().contains(k))
                .collect(Collectors.toList());
    }

    public Member get(String memberId) {
        return memberDao.get(memberId);
    }

    public Member findByPhoneNo(String phoneNo) {
        return memberDao.getByPhoneNo(phoneNo.trim());
    }

    /** สมัครสมาชิกด่วนหน้าร้าน · ข้อมูลผิดโยน IllegalArgumentException ให้หน้าจอแสดง */
    public Member add(Member m) {
        validate(m);
        if (memberDao.getByPhoneNo(m.getMemberPhoneNo()) != null) {
            throw new IllegalArgumentException("เบอร์ " + m.getMemberPhoneNo() + " เป็นสมาชิกอยู่แล้ว");
        }
        return memberDao.save(m);
    }

    public Member edit(Member m) {
        validate(m);
        Member samePhone = memberDao.getByPhoneNo(m.getMemberPhoneNo());
        if (isTakenByOther(samePhone, m)) {
            throw new IllegalArgumentException("เบอร์นี้เป็นของสมาชิก " + samePhone.getMemberId());
        }
        return memberDao.update(m);
    }

    public boolean remove(String memberId) {
        Member target = new Member();
        target.setMemberId(memberId);
        return memberDao.delete(target) > 0;
    }

    /** เรียกจาก PosService ตอนปิดบิล */
    public void addPoints(String memberId, double netAmount) {
        Member m = memberDao.get(memberId);
        if (m != null) {
            m.setPointBalance(m.getPointBalance() + (int) (netAmount / BAHT_PER_POINT));
            memberDao.update(m);
        }
    }

    private boolean isTakenByOther(Member found, Member editing) {
        return found != null && !found.getMemberId().equals(editing.getMemberId());
    }

    private void validate(Member m) {
        if (m.getMemberName() == null || m.getMemberName().isBlank()) {
            throw new IllegalArgumentException("กรุณากรอกชื่อสมาชิก");
        }
        if (m.getMemberPhoneNo() == null || !m.getMemberPhoneNo().matches("0\\d{9}")) {
            throw new IllegalArgumentException("เบอร์โทรต้องเป็นตัวเลข 10 หลัก ขึ้นต้นด้วย 0");
        }
    }
}

package com.example.bonussuki.service;

import com.example.bonussuki.dao.BillDao;
import com.example.bonussuki.dao.BillProductDao;
import com.example.bonussuki.dao.DiningTableDao;
import com.example.bonussuki.helper.DatabaseHelper;
import com.example.bonussuki.helper.DateUtil;
import com.example.bonussuki.model.Bill;
import com.example.bonussuki.model.BillLine;
import com.example.bonussuki.model.BillProduct;
import com.example.bonussuki.model.DiningTable;
import java.sql.Connection;
import java.sql.SQLException;
import java.util.List;

/** flow หน้าร้านทั้งหมด (เปิดโต๊ะ → สั่ง → ชำระเงิน) ใช้หลายตาราง จึงตั้งชื่อตามฟีเจอร์ ไม่ใช่ชื่อตาราง */
public class PosService {

    public static final String PAY_CASH = "เงินสด";
    public static final String PAY_QR = "QR Code";

    private final BillDao billDao = new BillDao();
    private final BillProductDao billProductDao = new BillProductDao();
    private final DiningTableDao diningTableDao = new DiningTableDao();
    private final PromotionService promotionService = new PromotionService();
    private final StockItemService stockItemService = new StockItemService();
    private final MemberService memberService = new MemberService();

    /** เปิดโต๊ะ: ต้องระบุจำนวนคน และโต๊ะต้องว่าง (กันเปิดโต๊ะซ้ำ) */
    public Bill openTable(String tableId, int amountMember) {
        DiningTable table = diningTableDao.get(tableId);
        if (table == null || !DiningTable.STATUS_VACANT.equals(table.getTableStatus())) {
            throw new IllegalStateException("โต๊ะนี้ไม่ว่าง");
        }
        if (amountMember < 1 || amountMember > table.getTableSize()) {
            throw new IllegalArgumentException("จำนวนลูกค้าต้องอยู่ระหว่าง 1–" + table.getTableSize() + " คน");
        }
        Bill bill = new Bill(tableId, amountMember, DateUtil.today(), DateUtil.nowTime());
        require(billDao.save(bill) != null, "สร้างบิลไม่สำเร็จ");
        diningTableDao.updateStatus(tableId, DiningTable.STATUS_IN_USE);
        return bill;
    }

    /** PK ของ Bill_Product คือ (Order_ID, Product_ID) → สั่งเมนูเดิมซ้ำ = แก้ Unit แถวเดิม ไม่ INSERT แถวใหม่ */
    public void addProduct(String orderId, String productId, int unit) {
        BillProduct line = billProductDao.get(orderId, productId);
        if (line == null && unit > 0) {
            billProductDao.save(new BillProduct(orderId, productId, DateUtil.now(), unit));
        } else if (line != null && line.getUnit() + unit <= 0) {
            billProductDao.delete(line);
        } else if (line != null) {
            line.setUnit(line.getUnit() + unit);
            billProductDao.update(line);
        }
    }

    public void removeProduct(String orderId, String productId) {
        BillProduct line = billProductDao.get(orderId, productId);
        if (line != null) {
            billProductDao.delete(line);
        }
    }

    public List<BillLine> getLines(String orderId) {
        return billProductDao.findLinesByOrderId(orderId);   // JOIN Product เอาชื่อ+ราคามาด้วย
    }

    public double calcTotal(List<BillLine> lines) {
        double total = 0;
        for (BillLine line : lines) {
            total += line.getLineTotal();
        }
        return round2(total);
    }

    /**
     * ชำระเงิน: ทุกขั้นอยู่ใน transaction เดียว พลาดขั้นไหนย้อนกลับทั้งหมด
     * เงินทอน = receivedAmount − ยอดสุทธิ (Derived ไม่เก็บลงตาราง)
     */
    public Bill checkout(String orderId, String paymentType, double receivedAmount,
            String promotionId, String memberId) throws SQLException {
        Connection conn = DatabaseHelper.getConnect();
        conn.setAutoCommit(false);
        try {
            Bill bill = billDao.get(orderId);
            double total = calcTotal(getLines(orderId));
            double discount = promotionService.calcDiscount(promotionId, bill, total);
            double net = round2(total - discount);
            double received = PAY_QR.equals(paymentType) ? net : round2(receivedAmount);
            if (received < net) {
                throw new IllegalArgumentException(String.format("รับเงินไม่พอ ขาดอีก %,.2f บาท", net - received));
            }
            bill.setTotalPrice(total);
            bill.setDiscountAmount(discount);
            bill.setReceivedAmount(received);
            bill.setPaymentType(paymentType);
            bill.setPaymentDate(DateUtil.today());
            bill.setEndTime(DateUtil.nowTime());
            bill.setPromotionId(promotionId);
            bill.setMemberId(memberId);
            if (promotionId != null) {
                bill.setPromoDateTime(DateUtil.now());
                bill.setPromoAmountMember(bill.getAmountMember());
            }
            require(billDao.update(bill) != null, "บันทึกบิลไม่สำเร็จ");
            stockItemService.deductByBom(orderId);              // ตัดสต็อกตามสูตร BOM
            if (memberId != null) {
                memberService.addPoints(memberId, net);
            }
            diningTableDao.updateStatus(bill.getTableId(), DiningTable.STATUS_VACANT);
            conn.commit();
            return bill;
        } catch (RuntimeException | SQLException ex) {
            conn.rollback();
            throw ex;
        } finally {
            conn.setAutoCommit(true);
        }
    }

    /** ปัดทศนิยม 2 ตำแหน่ง (TOR: เงินทอนแม่นยำระดับทศนิยม 2 ตำแหน่ง) */
    public static double round2(double value) {
        return Math.round(value * 100.0) / 100.0;
    }

    /** DAO กลืน SQLException แล้วคืน null → ต้องเช็คเอง ไม่งั้น rollback ไม่ทำงาน */
    private static void require(boolean ok, String message) {
        if (!ok) {
            throw new IllegalStateException(message);
        }
    }
}

public double calcDiscount(String promotionId, Bill bill, double total) {
    if (promotionId == null) {
        return 0;
    }
    Promotion p = promotionDao.get(promotionId);
    String today = DateUtil.today();
    if (today.compareTo(p.getStartDate()) < 0 || today.compareTo(p.getEndDate()) > 0) {
        throw new IllegalArgumentException("โปรโมชั่น " + p.getProName() + " ไม่อยู่ในช่วงวันที่ใช้ได้");
    }
    if (bill.getAmountMember() < p.getNumberPeople()) {
        throw new IllegalArgumentException("โปรนี้ต้องมาอย่างน้อย " + p.getNumberPeople() + " ท่าน");
    }
    // รูปเล่มไม่มีคอลัมน์บอกวิธีคิดส่วนลด จึงต้องเขียนตามรหัสโปร (ทีมต้องตกลงกัน ดูหัวข้อ 2)
    switch (p.getPromotionId()) {
        case "P001":                                   // มา 4 จ่าย 3: ลดค่าบุฟเฟต์ 1 ท่าน
            return cheapestBuffetPrice(bill.getOrderId());
        case "P002":                                   // สมาชิกลด 10%
            return PosService.round2(total * 0.10);
        case "P003":                                   // มา 2 ฟรีน้ำซุปพิเศษ 1 หม้อ
            return SPECIAL_SOUP_PRICE;
        default:
            return 0;
    }
